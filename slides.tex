\begin{frame}{1. Problem and dataset}
  \begin{itemize}
    \item bina.az sale listings (Kaggle, \texttt{house\_sale.csv}):
          100{,}775 rows, 51 columns, Azerbaijani names, scraped Oct 2024.
    \item \textbf{Task A --- regression:} predict the price; we model
          $\log(\text{price})$ because prices are heavy-tailed.
    \item \textbf{Task B --- classification:} price tier, premium if
          price $>$ median of the \emph{training} prices (208{,}000 AZN).
    \item Golden rule: tree and SVM are our own NumPy code;
          scikit-learn only as the baseline.
  \end{itemize}
\end{frame}

\begin{frame}{4. Our decision tree}
  \small
  \begin{itemize}
    \item Criteria: Gini $1-\sum_k p_k^2$, entropy $-\sum_k p_k\log_2 p_k$,
          MSE (node variance); gain $= I(\text{parent})-\sum_c \frac{N_c}{N} I(c)$.
    \item Split search: sort each feature once, cumulative sums give the
          impurity of \emph{every} threshold at once; threshold = midpoint.
    \item Stopping: \texttt{max\_depth}, \texttt{min\_samples\_split},
          \texttt{min\_samples\_leaf}, \texttt{min\_impurity\_decrease}.
    \item Leaves: majority class / mean; \texttt{predict\_proba} from leaf
          frequencies; impurity-decrease feature importances.
    \item Test: same splits as sklearn on a tiny fully-determined example.
  \end{itemize}
  \centering
  \includegraphics[width=0.75\textwidth]{tree_curves.png}
\end{frame}

\begin{frame}{5. Our SVM: Pegasos + random Fourier features}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \small
      \[
        J(\mathbf{w},b)=\tfrac{\lambda}{2}\lVert\mathbf{w}\rVert^2
        +\tfrac{1}{n}\textstyle\sum_i \max(0,1-y_i(\mathbf{w}^\top\mathbf{x}_i+b))
      \]
      \begin{itemize}
        \item $\eta_t = 1/(\lambda t)$; shrink $\mathbf{w}$, add
              $\eta_t y_i \mathbf{x}_i$ if $y_i f(\mathbf{x}_i)<1$.
        \item Bias folded in as a constant feature; mini-batches of 32.
        \item Features standardised with training statistics.
        \item Margin $2/\lVert\mathbf{w}\rVert$; support vectors
              $y_i f(\mathbf{x}_i)\le 1$.
        \item Non-linear: random Fourier features (RBF), then the same
              linear solver.
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\textwidth]{svm_curves.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{6. Experimental setup}
  \begin{itemize}
    \item Hyperparameters picked on the \textbf{validation} split only;
          the test set is used once.
    \item Task A metrics: RMSE, MAE, $R^2$ on $\log(\text{price})$ and in AZN.
    \item Task B metrics: precision, recall, F1, ROC-AUC, confusion matrix.
    \item Baselines with matched settings: \texttt{DecisionTreeRegressor},
          \texttt{DecisionTreeClassifier}, \texttt{SGDClassifier(hinge)}
          ($\alpha=\lambda$), \texttt{SVC(rbf)} ($C=1/(n\lambda)$),
          scaling inside a \texttt{Pipeline}.
    \item Uncertainty: 5-fold CV on train+val (threshold and scaling refit
          in every fold). Seed 42 everywhere.
  \end{itemize}
\end{frame}

    \textbf{Ours: RFF SVM}       & --- & --- & 0.892 & 0.964 & 0.878 $\pm$ 0.018 \\

    \bottomrule

  \vspace{0.8em}
  \begin{itemize}
    \item Our trees: same root split as sklearn, 967 vs 966 leaves (A),
          98.4\% identical predictions (B). Gaps: tie-breaking, float32
          thresholds, zero-gain splits.
    \item Linear SVM: $\lVert\mathbf{w}\rVert$ 3.63 vs 3.65, margin 0.55,
          33.9\% of training points are support vectors.
    \item Time: our tree 15--40$\times$ slower than sklearn; RFF SVM fits
          in 2~s vs 55~s for the exact SVC.
  \end{itemize}

      \small
      \textbf{Limitations}
      \begin{itemize}
        \item Asking prices from one scrape, not sale prices.
        \item Near-duplicates and data-entry errors remain.
        \item Only top-20 districts encoded; text not used.
        \item Our tree is much slower than sklearn's.
